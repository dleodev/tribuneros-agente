langgraph-checkpoint-postgres>=3.1,<4

# Con [binary], que trae la libpq adentro. Sin eso hay que instalar la
# libpq del sistema aparte, y el error que tira ("no pq wrapper available")
# no se parece en nada a lo que falta.
psycopg[binary]>=3.2,<4
